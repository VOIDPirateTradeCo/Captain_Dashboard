// SHA-256 hash function
async function sha256(str) {
    const encoder = new TextEncoder();
    const data = encoder.encode(str);
    const hashBuffer = await crypto.subtle.digest('SHA-256', data);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
}

// Add a log entry
function addLog(message, severity = 'info') {
    const log = document.getElementById('live-log');
    const now = new Date();
    const timeStr = `[${String(now.getHours()).padStart(2,'0')}:${String(now.getMinutes()).padStart(2,'0')}:${String(now.getSeconds()).padStart(2,'0')}]`;
    const colorClass = severity === 'error' ? 'offline' : severity === 'warn' ? 'warning' : 'online';
    const entry = document.createElement('div');
    entry.className = 'log-entry new';
    entry.innerHTML = `<span class="time">${timeStr}</span> <span class="${colorClass}">${message}</span>`;
    log.insertBefore(entry, log.firstChild);
    // Keep only last 20 entries
    while (log.children.length > 20) {
        log.removeChild(log.lastChild);
    }
}

// WHITE WHALE passphrase unlock
async function unlockWhale() {
    const input = document.getElementById('whale-passphrase');
    const error = document.getElementById('whale-error');
    const passphrase = input.value;

    if (!passphrase) {
        error.textContent = 'Enter passphrase to unlock';
        error.className = 'whale-message whale-error-msg';
        return;
    }

    error.textContent = '🔐 Authenticating...';
    error.className = 'whale-message';

    const hash = await sha256(passphrase);

    try {
        const resp = await fetch(`${WHALE_API}?passphrase_hash=${hash}&threat_detected=true`, {cache: 'no-store'});
        const data = await resp.json();

        if (data.error) {
            error.textContent = '❌ ' + data.error;
            error.className = 'whale-message whale-error-msg';
            addLog(`WHITE WHALE unlock ATTEMPT FAILED`, 'error');
            return;
        }

        if (data.white_whale_hash_verified) {
            document.body.classList.add('whale-unlocked');
            error.textContent = '✅ WHITE WHALE protocols unlocked!';
            error.className = 'whale-message whale-success';
            addLog(`WHITE WHALE PROTOCOLS UNLOCKED by Captain`, 'info');

            // Populate WHITE WHALE protocol steps
            const wp = data.white_whale_protocol || {};
            setText('whale-step-1', `${wp.step_1_vault_lockdown ? wp.step_1_vault_lockdown.status : '—'}${wp.step_1_vault_lockdown ? ' — ' + wp.step_1_vault_lockdown.detail : ''}`, wp.step_1_vault_lockdown && wp.step_1_vault_lockdown.status === 'PASS' ? 'online' : 'offline');
            setText('whale-step-2', `${wp.step_2_encrypted_channel ? wp.step_2_encrypted_channel.status : '—'}${wp.step_2_encrypted_channel ? ' — ' + wp.step_2_encrypted_channel.detail : ''}`, wp.step_2_encrypted_channel && wp.step_2_encrypted_channel.status === 'ESTABLISHED' ? 'online' : 'offline');
            setText('whale-step-3', `${wp.step_3_no_secrets_git ? wp.step_3_no_secrets_git.status : '—'}${wp.step_3_no_secrets_git ? ' — ' + wp.step_3_no_secrets_git.detail : ''}`, wp.step_3_no_secrets_git && wp.step_3_no_secrets_git.status === 'PASS' ? 'online' : 'offline');
            setText('whale-unlocked-at', data.unlocked_at || 'unknown');

            // BLACK WHALE
            const bw = data.black_whale || {};
            setText('black-whale-threats', bw.threats_detected || 0, bw.threats_detected === 0 ? 'online' : 'offline');
            setText('black-whale-status', bw.status || 'UNKNOWN', bw.threats_detected === 0 ? 'online' : 'offline');
            setText('black-whale-total', bw.total_devices || 0);

            // GREEN WHALE
            const gw = data.green_whale || {};
            setText('green-whale-status', gw.status || 'UNKNOWN', gw.status && gw.status === 'HEALTHY' ? 'online' : 'offline');
            const checks = gw.checks || {};
            const passing = Object.values(checks).filter(v => v).length;
            const total = Object.keys(checks).length;
            setText('green-whale-checks', `${passing}/${total} checks passing`, passing === total ? 'online' : 'offline');
            setText('green-whale-maint', gw.maintenance_needed ? '⚠️ YES' : '✅ NO', gw.maintenance_needed ? 'offline' : 'online');

            // GREEN WHALE port checks
            for (const [label, elId] of [['npm_port_81', 'gw-port-81'], ['nginx_port_80', 'gw-port-80'],
                                         ['docker_api', 'gw-port-2376'], ['health_check', 'gw-port-9999'],
                                         ['dashboard', 'gw-port-8080'], ['smb_vault', 'gw-port-445']]) {
                const up = checks[label];
                setText(elId, up ? '🟢 UP' : '🔴 DOWN', up ? 'online' : 'offline');
            }

            // Unknown devices list
            const unknownList = document.getElementById('unknown-devices-list');
            const unknowns = bw.unknown_devices || [];
            if (unknowns.length === 0) {
                unknownList.innerHTML = '<div class="online">✅ No unknown devices detected — network is secure</div>';
            } else {
                unknownList.innerHTML = unknowns.map(d =>
                    `<div class="status-item"><span class="label">${d.ip}</span><span class="value">${d.hostname || 'Unknown device'}</span></div>`
                ).join('');
            }
        } else {
            error.textContent = '❌ Invalid passphrase';
            error.className = 'whale-message whale-error-msg';
            addLog(`WHITE WHALE unlock FAILED — invalid passphrase`, 'error');
        }
    } catch(e) {
        error.textContent = '❌ Network error — check connection';
        error.className = 'whale-message whale-error-msg';
        addLog(`WHITE WHALE unlock error: ${e.message}`, 'error');
    }
}

function setText(id, text, cls = '') {
    const el = document.getElementById(id);
    if (!el) return;
    el.innerHTML = cls ? `<span class="${cls}">${text}</span>` : text;
}

function setPorts(id, ports) {
    const el = document.getElementById(id);
    if (!el) return;
    if (!ports || ports.length === 0) {
        el.innerHTML = '<span style="color:#666;">—</span>';
        return;
    }
    el.innerHTML = ports.slice(0, 6).map(p => `<span class="port-tag">${p}</span>`).join('') +
        (ports.length > 6 ? `<span class="port-tag">+${ports.length - 6}</span>` : '');
}

function setShipStatus(id, status) {
    const el = document.getElementById(id);
    if (!el) return;
    if (status === 'online') {
        el.innerHTML = '<span class="ship-online">🟢 ONLINE</span>';
    } else {
        el.innerHTML = '<span class="ship-offline">🔴 OFFLINE</span>';
    }
}

function applyGithub(github) {
  const el = document.getElementById('github-issues-list');
  if (!el) return;
  const repo = github?.repo || '';
  const issues = github?.issues || [];
  const fallback = github?.source === 'github_fallback';
  const error = fallback ? (github?.error || 'GitHub feed unavailable') : '';
  if (!repo) {
    el.innerHTML = '<div class="terminal-scan">GitHub repo not configured</div>';
    return;
  }
  const header = `<div class="status-item"><span class="label">Repo</span><span class="value">${repo}${error ? ` — ${error}` : ''}</span></div>`;
  if (!issues.length) {
    el.innerHTML = `${header}<div class="terminal-scan">No open issues to display</div>`;
    return;
  }
  el.innerHTML = header + issues.map(i => {
    const labels = (i.labels || []).map(l => `<span class="port-tag">${l}</span>`).join(' ');
    return `<div class="status-item"><span class="label">#${i.number}</span><span class="value">${i.title} ${labels} <a href="${i.url}" target="_blank" class="metric">Open</a></span></div>`;
  }).join('');
}

async function applyFleetConnectivity() {
    try {
        const r = await fetch(FLEET_CONNECTIVITY_API, {cache: 'no-store'});
        if (!r.ok) return;
        const data = await r.json();
        const nodes = data?.nodes?.nodes || data?.nodes || [];
        const tbody = document.getElementById('fleet-connectivity-body');
        if (!tbody) return;
        const knownIps = new Set((data?.nodes?.nodes || []).map(n => n.ip));
        const rows = nodes.map((n, i) => {
            const status = n.online ? 'online' : 'offline';
            const tag = n.online ? '🟢 ONLINE' : '🔴 OFFLINE';
            return `<tr><td>${i+1}</td><td>${n.name}</td><td>${n.ip}</td><td class="${status}">${tag}</td></tr>`;
        }).join('');
        tbody.innerHTML = rows || '<tr><td colspan="4">No fleet data</td></tr>';
        const summary = document.getElementById('fleet-connectivity-summary');
        if (summary) summary.textContent = `${nodes.length} node(s)`;
    } catch (e) {}
}

async function applyQueueStatus() {

/* ==================== FODAVP BUTTON HANDLERS ==================== */
function activateFodavp() {
    const btn = document.getElementById('fodavpActivateBtn');
    const status = document.getElementById('fodavpStatus');
    const detail = document.getElementById('fodavpDetail');
    
    btn.disabled = true;
    btn.textContent = '⚡ ACTIVATING...';
    status.textContent = '🔵 ACTIVATING';
    detail.textContent = 'Starting FODAVP engine...';
    
    fetch('/api/fodavp/trigger', {method: 'POST'})
        .then(response => response.json())
        .then(data => {
            status.textContent = '🟢 RUNNING';
            detail.textContent = 'Cycle started: ' + new Date().toLocaleTimeString();
            setTimeout(updateFodavpStatus, 5000);
            window.fodavpPoll = setInterval(updateFodavpStatus, 30000);
            setTimeout(() => {
                btn.disabled = false;
                btn.textContent = '⚔️ ACTIVATE FODAVP PROTOCAL ⚡';
            }, 3000);
        })
        .catch(err => {
            status.textContent = '🔴 ERROR';
            detail.textContent = 'Activation failed: ' + err.message;
            btn.disabled = false;
            btn.textContent = '⚔️ ACTIVATE FODAVP PROTOCAL ⚡';
        });
}

function stopFodavp() {
    if (!confirm('🛑 STOP ALL FODAVP SYSTEMS?\nThis will terminate the fleet automation engine immediately.')) return;
    
    fetch('/api/stop-fodavp', {method: 'POST'})
        .then(response => response.json())
        .then(data => {
            document.getElementById('fodavpStatus').textContent = '🟡 STOPPED';
            document.getElementById('fodavpDetail').textContent = 'Stopped: ' + new Date().toLocaleTimeString();
            clearInterval(window.fodavpPoll);
        })
        .catch(err => {
            console.error('Stop failed:', err);
        });
}

function updateFodavpStatus() {
    fetch('/api/fodavp-status')
        .then(response => response.json())
        .then(data => {
            document.getElementById('fodavpEvidence').textContent = (data.evidence_count || 0) + ' files';
            document.getElementById('fodavpFleet').textContent = 
                'SQUIDSTATION: ' + (data.squid || '🟡') + ' | PINKCADY: ' + (data.pink || '🟡') + ' | STEALTH: ' + (data.stealth || '🟡');
            document.getElementById('fodavpRate').textContent = data.cycle_rate || '~30s/cycle';
            if (data.running) {
                document.getElementById('fodavpStatus').textContent = '🟢 RUNNING';
            }
        })
        .catch(err => {
            console.warn('Status check failed:', err);
        });
}

function loadSettings() {
    fetch('/api/settings')
        .then(r => r.ok ? r.json() : Promise.reject(r.status))
        .then(data => {
            if (data.error) return;
            const set = (id, val) => { const el = document.getElementById(id); if (!el) return; if (el.type === 'checkbox') el.checked = !!val; else el.value = val ?? ''; };
            set('setting-monitoring_interval', data.monitoring_interval);
            set('setting-alert_threshold', data.alert_threshold);
            set('setting-theme', data.theme);
            set('setting-auto_refresh', data.auto_refresh);
            set('setting-fleet_monitor', data.fleet_monitor);
            set('setting-offline_queue', data.offline_queue);
        })
        .catch(err => console.warn('Settings load failed:', err));
}

function saveSettings() {
    const payload = {
        monitoring_interval: parseInt(document.getElementById('setting-monitoring_interval')?.value || '10', 10),
        alert_threshold: parseInt(document.getElementById('setting-alert_threshold')?.value || '3', 10),
        theme: document.getElementById('setting-theme')?.value || 'void',
        auto_refresh: document.getElementById('setting-auto_refresh')?.checked ?? true,
        fleet_monitor: document.getElementById('setting-fleet_monitor')?.checked ?? true,
        offline_queue: document.getElementById('setting-offline_queue')?.checked ?? false,
    };
    const status = document.getElementById('settings-status');
    if (status) status.textContent = 'Saving...';
    fetch('/api/settings', {method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(payload)})
        .then(r => r.ok ? r.json() : Promise.reject(r.status))
        .then(data => {
            if (status) status.textContent = data.ok ? 'Saved' : 'Save failed';
        })
        .catch(err => {
            if (status) status.textContent = 'Save failed';
            console.warn('Settings save failed:', err);
        });
}
